name: Build FreeWallpaper IPA

on:
  workflow_dispatch:
  push:
    branches: [ main, master ]

jobs:
  build:
    runs-on: macos-15   # iOS 17+ 目标；如需液态玻璃 SDK 用带 Xcode 26+ 的 image（如 macos-latest）
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Select Xcode
        run: |
          ls /Applications | grep Xcode || true
          sudo xcode-select -s /Applications/Xcode.app/Contents/Developer

      - name: Archive (unsigned)
        run: |
          xcodebuild archive \
            -project FreeWallpaperiOS.xcodeproj \
            -scheme FreeWallpaperiOS \
            -configuration Release \
            -destination 'generic/platform=iOS' \
            -archivePath build/FreeWallpaperiOS.xcarchive \
            CODE_SIGNING_ALLOWED=NO \
            SKIP_INSTALL=NO

      - name: Pack IPA
        run: |
          mkdir -p build/Payload
          cp -R build/FreeWallpaperiOS.xcarchive/Products/Applications/FreeWallpaperiOS.app build/Payload/
          cd build && zip -qry FreeWallpaperiOS.ipa Payload

      - name: Upload IPA
        uses: actions/upload-artifact@v4
        with:
          name: FreeWallpaperiOS-IPA
          path: build/FreeWallpaperiOS.ipa
          if-no-files-found: error
